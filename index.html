guitarix (kleag) apt repository

Suites: bookworm resolute 

sudo install -d /etc/apt/keyrings
sudo curl -fsSLo /etc/apt/keyrings/kleag-guitarix.asc https://kleag.github.io/guitarix/kleag-guitarix.asc
printf 'Types: deb\nURIs: https://kleag.github.io/guitarix\nSuites: %s\nComponents: main\nSigned-By: /etc/apt/keyrings/kleag-guitarix.asc\n' \
  "$(. /etc/os-release; echo $VERSION_CODENAME)" | sudo tee /etc/apt/sources.list.d/kleag-guitarix.sources
sudo apt update
sudo apt install guitarix-kleag
